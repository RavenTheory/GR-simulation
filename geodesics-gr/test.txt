import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from matplotlib.animation import FuncAnimation


data = pd.read_csv("trajectory.csv", sep=";")

# Spherical -> Cartesian for every particle at every frame
r = data["r"].to_numpy()
theta = data["theta"].to_numpy()
phi = data["phi"].to_numpy()

data["x"] = r * np.sin(theta) * np.cos(phi)
data["y"] = r * np.sin(theta) * np.sin(phi)
data["z"] = r * np.cos(theta)


fig = plt.figure()
ax = fig.add_subplot(projection="3d")

limit = np.max(np.abs(data["x"])) * 1.3

#other limit_
#limit = 12

ax.set_xlim(-limit, limit)
ax.set_ylim(-limit, limit)
ax.set_zlim(-limit, limit)
ax.set_box_aspect((1, 1, 1), zoom=1.4)


# Event horizon
M = 1.0
r_h = 2 * M

theta_s = np.linspace(0, np.pi, 50)
phi_s = np.linspace(0, 2*np.pi, 50)
theta_s, phi_s = np.meshgrid(theta_s, phi_s)

i = r_h * np.sin(theta_s) * np.cos(phi_s)
j = r_h * np.sin(theta_s) * np.sin(phi_s)
k = r_h * np.cos(theta_s)

ax.plot_surface(i, j, k, color="black")


# All particles
particles, = ax.plot(
    [], [], [],
    linestyle="",
    marker="o",
    color="red"
)


def animate(frame_number):
    frame = data[data["frame"] == frame_number]

    particles.set_data(
        frame["x"].to_numpy(),
        frame["y"].to_numpy()
    )

    particles.set_3d_properties(
        frame["z"].to_numpy()
    )

    
    return particles,


frames = data["frame"].unique()

fps = 120
skip = 10

ani = FuncAnimation(
    fig,
    animate,
    frames=frames[::skip],
    interval=1000 / fps
)

ax.set_xlabel("x")
ax.set_ylabel("y")
ax.set_zlabel("z")

ani.save("geodesics.mp4", writer="ffmpeg", fps=fps)

plt.show()
